Module Opentelemetry_trace.Extensions

The extension events for Trace_core.

type Otrace.span +=
  1. | Span_otel of OTEL.Span.t
    (*

    The type of span used for OTEL

    *)
type Otrace.extension_event +=
  1. | Ev_link_span of Otrace.span * OTEL.Span_ctx.t
    (*

    Link the given span to the given context. The context isn't the parent, but the link can be used to correlate both spans.

    *)
  2. | Ev_record_exn of {
    1. sp : Otrace.span;
    2. exn : exn;
    3. bt : Stdlib.Printexc.raw_backtrace;
    }
    (*

    Record exception and potentially turn span to an error

    *)
  3. | Ev_set_span_kind of Otrace.span * OTEL.Span_kind.t
  4. | Ev_set_span_status of Otrace.span * OTEL.Span_status.t
type Otrace.metric +=
  1. | Metric_hist of OTEL.Metrics.histogram_data_point

  3. | Metric_sum_float of float
